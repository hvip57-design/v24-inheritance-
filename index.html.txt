const HEIRS = [
  { id: 'husband', name: 'زوج', count: false },
  { id: 'wife', name: 'زوجة', count: true, max: 4 },
  { id: 'father', name: 'أب', count: false },
  { id: 'mother', name: 'أم', count: false },
  { id: 'son', name: 'ابن', count: true, max: 10 },
  { id: 'daughter', name: 'بنت', count: true, max: 10 },
  { id: 'brothers', name: 'أخ', count: true, max: 10 },
  { id: 'sisters', name: 'أخت', count: true, max: 10 }
];

function buildForm() {
  const grid = document.getElementById('heirsGrid');
  grid.innerHTML = '';
  HEIRS.forEach(h => {
    const div = document.createElement('div');
    div.className = 'heir-item';
    div.innerHTML = `
      <input type="checkbox" id="chk_${h.id}" onchange="onCheck('${h.id}')">
      <label for="chk_${h.id}">${h.name}</label>
      ${h.count ? `<input type="number" id="num_${h.id}" min="1" max="${h.max}" value="1" style="display:none">` : ''}
    `;
    grid.appendChild(div);
  });
}

function onCheck(id) {
  const chk = document.getElementById('chk_' + id);
  const num = document.getElementById('num_' + id);
  if (num) num.style.display = chk.checked ? 'block' : 'none';
  if (id === 'husband' && chk.checked) { document.getElementById('chk_wife').checked = false; document.getElementById('num_wife').style.display = 'none'; }
  if (id === 'wife' && chk.checked) { document.getElementById('chk_husband').checked = false; }
}

function resetAll() {
  HEIRS.forEach(h => {
    document.getElementById('chk_' + h.id).checked = false;
    const num = document.getElementById('num_' + h.id);
    if (num) { num.value = 1; num.style.display = 'none'; }
  });
  document.getElementById('warningMsg').style.display = 'none';
  document.getElementById('resultSection').style.display = 'none';
}

function gcd(a, b) { return b ? gcd(b, a % b) : a; }
function F(n, d=1) { if (d < 0) { n = -n; d = -d; } const g = gcd(Math.abs(n), d) || 1; return { n: n/g, d: d/g }; }
function add(a, b) { return F(a.n*b.d + b.n*a.d, a.d*b.d); }
function sub(a, b) { return F(a.n*b.d - b.n*a.d, a.d*b.d); }
function mul(a, b) { return F(a.n*b.n, a.d*b.d); }
function div(a, b) { return F(a.n*b.d, a.d*b.n); }
function toStr(a) { return a.d === 1 ? a.n.toString() : a.n + '/' + a.d; }
function toPct(a) { return (a.n / a.d * 100).toFixed(2) + '%'; }

function calc(h) {
  const son = h.son || 0, dau = h.daughter || 0;
  const hasChild = son > 0 || dau > 0;
  const hasMaleChild = son > 0;
  const father = !!h.father, mother = !!h.mother;
  const kallah = !father && !hasMaleChild;
  const brothers = h.brothers || 0, sisters = h.sisters || 0;
  const totalSib = brothers + sisters;
  const husband = !!h.husband, wife = h.wife || 0;

  let count = 0;
  if (husband) count++;
  if (wife) count++;
  if (father) count++;
  if (mother) count++;
  if (son) count += son;
  if (dau) count += dau;
  if (brothers) count += brothers;
  if (sisters) count += sisters;

  if (count === 1) {
    const res = {};
    if (husband) res['زوج'] = { frac: F(1), waa: '-', type: 'الوحيد' };
    if (wife) res['زوجة'] = { frac: F(1), waa: '-', type: 'الوحيد' };
    if (father) res['أب'] = { frac: F(1), waa: '-', type: 'الوحيد' };
    if (mother) res['أم'] = { frac: F(1), waa: '-', type: 'الوحيد' };
    if (son) res['ابن'] = { frac: F(1), waa: '-', type: 'الوحيد' };
    if (dau) res['بنت'] = { frac: F(1), waa: '-', type: 'الوحيد' };
    if (brothers) res['أخ'] = { frac: F(1), waa: '-', type: 'الوحيد' };
    if (sisters) res['أخت'] = { frac: F(1), waa: '-', type: 'الوحيد' };
    return res;
  }

  const K = {}, Fsh = {};

  if (husband && hasChild) K['زوج'] = { frac: F(1,4), type: 'ثابت' };
  if (wife && hasChild) K['زوجة'] = { frac: F(1,8), type: 'ثابت' };
  if (mother && totalSib >= 2 && kallah) K['أم'] = { frac: F(1,6), type: 'ثابت' };
  
  if (kallah && totalSib >= 3 && dau > 0 && son === 0) {
    if (brothers > 0 && sisters > 0) {
      const tp = 2*brothers + sisters;
      K['إخوة'] = { frac: mul(F(2*brothers, tp), F(1,3)), type: 'ثابت' };
      K['أخوات'] = { frac: mul(F(sisters, tp), F(1,3)), type: 'ثابت' };
    } else if (brothers >= 3) {
      K['إخوة'] = { frac: F(1,3), type: 'ثابت' };
    } else if (sisters >= 3) {
      K['أخوات'] = { frac: F(1,3), type: 'ثابت' };
    }
  }

  let Ksum = F(0);
  for (const k in K) Ksum = add(Ksum, K[k].frac);

  if (father && !hasChild) {
    if (mother) Fsh['أم'] = { frac: mul(F(1,3), sub(F(1), Ksum)), type: 'مفتوح' };
    let fsum = F(0);
    for (const k in Fsh) fsum = add(fsum, Fsh[k].frac);
    Fsh['أب'] = { frac: sub(sub(F(1), Ksum), fsum), type: 'عصبة' };
    return merge(K, Fsh);
  }

  const Fsize = sub(F(1), Ksum);

  if (husband && !hasChild) Fsh['زوج'] = { frac: mul(F(1,2), Fsize), type: 'تراكمي' };
  if (wife && !hasChild) Fsh['زوجة'] = { frac: mul(F(1,4), Fsize), type: 'ثابت' };
  if (father && hasChild) Fsh['أب'] = { frac: mul(F(1,6), Fsize), type: 'ثابت' };
  if (mother && !K['أم']) {
    if (hasChild) Fsh['أم'] = { frac: mul(F(1,6), Fsize), type: 'ثابت' };
    else Fsh['أم'] = { frac: mul(F(1,3), Fsize), type: 'مفتوح' };
  }

  if (son > 0) {
    let rem = Fsize;
    for (const k in Fsh) rem = sub(rem, Fsh[k].frac);
    if (dau > 0) {
      const tp = 2*son + dau;
      Fsh['ابن'] = { frac: mul(F(2*son, tp), rem), type: 'مفتوح' };
      Fsh['بنت'] = { frac: mul(F(dau, tp), rem), type: 'مفتوح' };
    } else Fsh['ابن'] = { frac: rem, type: 'مفتوح' };
  } else if (dau > 0) {
    Fsh['بنت'] = { frac: mul(dau === 1 ? F(1,2) : F(2,3), Fsize), type: dau === 1 ? 'ثابت' : 'تراكمي' };
  }

  if (kallah && son === 0 && totalSib > 0) {
    if (dau > 0) {
      if (totalSib === 1) {
        if (brothers === 1) Fsh['أخ'] = { frac: mul(F(1,6), Fsize), type: 'مفتوح' };
        else Fsh['أخت'] = { frac: mul(F(1,6), Fsize), type: 'مفتوح' };
      } else if (totalSib === 2) {
        if (brothers === 2) Fsh['إخوة'] = { frac: mul(F(1,3), Fsize), type: 'مفتوح' };
        else if (sisters === 2) Fsh['أخوات'] = { frac: mul(F(1,3), Fsize), type: 'مفتوح' };
        else { Fsh['إخوة'] = { frac: mul(F(1,6), Fsize), type: 'مفتوح' }; Fsh['أخوات'] = { frac: mul(F(1,6), Fsize), type: 'مفتوح' }; }
      }
    } else {
      if (totalSib === 1) {
        if (brothers === 1) {
          let rem = Fsize;
          for (const k in Fsh) rem = sub(rem, Fsh[k].frac);
          Fsh['أخ'] = { frac: rem, type: 'مفتوح' };
        } else Fsh['أخت'] = { frac: mul(F(1,2), Fsize), type: 'تراكمي' };
      } else {
        if (brothers > 0 && sisters > 0) {
          let rem = Fsize;
          for (const k in Fsh) rem = sub(rem, Fsh[k].frac);
          const tp = 2*brothers + sisters;
          Fsh['إخوة'] = { frac: mul(F(2*brothers, tp), rem), type: 'مفتوح' };
          Fsh['أخوات'] = { frac: mul(F(sisters, tp), rem), type: 'مفتوح' };
        } else if (sisters >= 2) {
          Fsh['أخوات'] = { frac: mul(F(2,3), Fsize), type: 'ثابت' };
        } else if (brothers >= 2) {
          let rem = Fsize;
          for (const k in Fsh) rem = sub(rem, Fsh[k].frac);
          Fsh['إخوة'] = { frac: rem, type: 'مفتوح' };
        }
      }
    }
  }

  let Fsum = F(0);
  for (const k in Fsh) Fsum = add(Fsum, Fsh[k].frac);
  let leftover = sub(Fsize, Fsum);

  if (leftover.n > 0) {
    const tarkumi = [], maftouh = [], thabit = [];
    for (const k in Fsh) {
      if (Fsh[k].type === 'تراكمي') tarkumi.push(k);
      else if (Fsh[k].type === 'مفتوح') maftouh.push(k);
      else if (Fsh[k].type === 'ثابت') thabit.push(k);
    }
    const thabitFiltered = thabit.filter(k => k !== 'زوجة' && k !== 'زوج');
    let eligible = tarkumi.length > 0 ? tarkumi : (maftouh.length > 0 ? maftouh : thabitFiltered);
    if (eligible.length > 0) {
      let tot = F(0);
      for (const k of eligible) tot = add(tot, Fsh[k].frac);
      for (const k of eligible) {
        Fsh[k].frac = add(Fsh[k].frac, mul(leftover, div(Fsh[k].frac, tot)));
      }
    }
  }

  return merge(K, Fsh);
}

function merge(K, Fsh) {
  const res = {};
  for (const k in K) res[k] = { frac: K[k].frac, waa: 'K', type: K[k].type };
  for (const k in Fsh) {
    if (res[k]) res[k].frac = add(res[k].frac, Fsh[k].frac);
    else res[k] = { frac: Fsh[k].frac, waa: 'F', type: Fsh[k].type };
  }
  return res;
}

function splitToIndividuals(data, h) {
  const rows = [];
  if (data['زوج']) rows.push({ name: 'زوج', frac: data['زوج'].frac, waa: data['زوج'].waa || 'F', type: data['زوج'].type });
  if (data['زوجة'] && h.wife > 0) {
    const each = div(data['زوجة'].frac, F(h.wife));
    for (let i = 1; i <= h.wife; i++) rows.push({ name: `زوجة ${i}`, frac: each, waa: data['زوجة'].waa || 'F', type: data['زوجة'].type });
  }
  if (data['أب']) rows.push({ name: 'أب', frac: data['أب'].frac, waa: data['أب'].waa || 'F', type: data['أب'].type });
  if (data['أم']) rows.push({ name: 'أم', frac: data['أم'].frac, waa: data['أم'].waa || 'F', type: data['أم'].type });
  if (data['ابن'] && h.son > 0) {
    const each = div(data['ابن'].frac, F(h.son));
    for (let i = 1; i <= h.son; i++) rows.push({ name: `ابن ${i}`, frac: each, waa: data['ابن'].waa || 'F', type: data['ابن'].type });
  }
  if (data['بنت'] && h.daughter > 0) {
    const each = div(data['بنت'].frac, F(h.daughter));
    for (let i = 1; i <= h.daughter; i++) rows.push({ name: `بنت ${i}`, frac: each, waa: data['بنت'].waa || 'F', type: data['بنت'].type });
  }
  if (data['إخوة'] && h.brothers > 0) {
    const each = div(data['إخوة'].frac, F(h.brothers));
    for (let i = 1; i <= h.brothers; i++) rows.push({ name: `أخ ${i}`, frac: each, waa: data['إخوة'].waa || 'F', type: data['إخوة'].type });
  }
  if (data['أخوات'] && h.sisters > 0) {
    const each = div(data['أخوات'].frac, F(h.sisters));
    for (let i = 1; i <= h.sisters; i++) rows.push({ name: `أخت ${i}`, frac: each, waa: data['أخوات'].waa || 'F', type: data['أخوات'].type });
  }
  if (data['أخ'] && h.brothers === 1 && !data['إخوة']) rows.push({ name: 'أخ', frac: data['أخ'].frac, waa: data['أخ'].waa || 'F', type: data['أخ'].type });
  if (data['أخت'] && h.sisters === 1 && !data['أخوات']) rows.push({ name: 'أخت', frac: data['أخت'].frac, waa: data['أخت'].waa || 'F', type: data['أخت'].type });
  return rows;
}

function calculate() {
  const h = {};
  HEIRS.forEach(x => {
    const chk = document.getElementById('chk_' + x.id);
    if (chk.checked) {
      const num = document.getElementById('num_' + x.id);
      h[x.id] = num ? parseInt(num.value) || 1 : 1;
    }
  });

  if (h.husband && h.wife) {
    document.getElementById('warningMsg').style.display = 'block';
    document.getElementById('resultSection').style.display = 'block';
    document.getElementById('resultContent').innerHTML = '<div class="error">❌ لا يمكن أن يجتمع الزوج والزوجة في الميراث.</div>';
    return;
  }

  if (Object.keys(h).length === 0) {
    document.getElementById('resultSection').style.display = 'block';
    document.getElementById('resultContent').innerHTML = '<div class="error">اختر وارثًا واحدًا على الأقل</div>';
    return;
  }

  document.getElementById('warningMsg').style.display = 'none';
  const estate = parseFloat(document.getElementById('estate').value) || 120;
  const res = calc(h);
  const individuals = splitToIndividuals(res, h);
  showResult(individuals, estate);
}

function showResult(rows, estate) {
  const sec = document.getElementById('resultSection');
  const con = document.getElementById('resultContent');
  sec.style.display = 'block';
  let html = '<table><thead><tr><th>الوارث</th><th>الوعاء</th><th>النوع</th><th>الكسر</th><th>النسبة</th><th>المبلغ</th></tr></thead><tbody>';
  let sum = F(0);
  rows.forEach(r => {
    sum = add(sum, r.frac);
    const amount = (r.frac.n / r.frac.d) * estate;
    let badgeClass = '';
    if (r.type === 'تراكمي') badgeClass = 'badge-tarkumi';
    else if (r.type === 'ثابت') badgeClass = 'badge-thabit';
    else if (r.type === 'مفتوح') badgeClass = 'badge-maftouh';
    else badgeClass = 'badge-thabit';
    let waaBadge = 'badge-F';
    if (r.waa === 'K') waaBadge = 'badge-K';
    html += `<tr><td><strong>${r.name}</strong></td><td><span class="badge ${waaBadge}">${r.waa}</span></td><td><span class="badge ${badgeClass}">${r.type}</span></td><td>${toStr(r.frac)}</td><td>${toPct(r.frac)}</td><td><strong>${amount.toFixed(2)}</strong></td></tr>`;
  });
  html += '</tbody></table>';
  html += `<div class="result-total">المجموع: <strong>${toStr(sum)}</strong> = ${toPct(sum)} = <strong>${estate.toFixed(2)}</strong></div>`;
  con.innerHTML = html;
}
